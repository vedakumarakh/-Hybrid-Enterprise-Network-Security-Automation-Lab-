! =========================================================
! vIOS1 - Site A edge router (BGP AS 65001)
! HOW TO APPLY:
!   1. Console into vIOS1 in EVE-NG (right-click node > console)
!   2. Type: enable        (no password by default on fresh vIOS)
!   3. Type: configure terminal
!   4. Paste everything below this line, then type: end
!   5. Save with: write memory
! =========================================================

hostname vIOS1
no ip domain-lookup
ip domain-name lab.local
!
enable secret Cisco@12345
username admin privilege 15 secret Cisco@12345
!
crypto key generate rsa modulus 1024
ip ssh version 2
!
line vty 0 4
 login local
 transport input ssh telnet
line con 0
 logging synchronous
!
interface Loopback0
 ip address 1.1.1.1 255.255.255.255
!
interface GigabitEthernet0/0
 description LINK-to-vIOS3-CORE
 ip address 10.0.12.1 255.255.255.252
 no shutdown
!
interface GigabitEthernet0/1
 description LINK-to-ASAv-outside
 ip address 10.10.1.1 255.255.255.252
 no shutdown
!
router ospf 1
 router-id 1.1.1.1
 network 10.10.1.0 0.0.0.3 area 0
 redistribute bgp 65001 subnets
!
router bgp 65001
 bgp router-id 1.1.1.1
 bgp log-neighbor-changes
 neighbor 10.0.12.2 remote-as 65000
 neighbor 10.0.12.2 description eBGP-to-CORE-vIOS3
 redistribute ospf 1
!
banner motd # vIOS1 - Site A Edge Router - Lab Environment #
!
end
write memory
