! =========================================================
! FortiGate - Site B firewall
! HOW TO APPLY:
!   1. Console into the FortiGate node in EVE-NG (or SSH once port1 has an IP)
!   2. Login: admin / (blank password on first boot) - it will force you
!      to set a new admin password immediately. Set it, then re-login.
!   3. Paste the blocks below into the CLI console one "config...end" block
!      at a time (FortiOS CLI is picky about pasting huge blocks at once -
!      do it section by section if paste fails).
! =========================================================

config system global
    set hostname FortiGate-SiteB
end

config system interface
    edit "port1"
        set alias "WAN-to-vIOS2"
        set ip 10.20.1.2 255.255.255.252
        set allowaccess ping https ssh
    next
    edit "port2"
        set alias "LAN-trunk-to-Switch5"
        set allowaccess ping
    next
    edit "port2.40"
        set alias "VLAN40-VPC9"
        set vdom "root"
        set ip 172.16.40.1 255.255.255.0
        set interface "port2"
        set vlanid 40
        set allowaccess ping https
    next
    edit "port2.50"
        set alias "VLAN50-Win"
        set vdom "root"
        set ip 172.16.50.1 255.255.255.0
        set interface "port2"
        set vlanid 50
        set allowaccess ping https
    next
end

config router ospf
    set router-id 10.20.1.2
    config area
        edit 0.0.0.0
        next
    end
    config network
        edit 1
            set prefix 10.20.1.0 255.255.255.252
            set area 0.0.0.0
        next
        edit 2
            set prefix 172.16.40.0 255.255.255.0
            set area 0.0.0.0
        next
        edit 3
            set prefix 172.16.50.0 255.255.255.0
            set area 0.0.0.0
        next
    end
end

! ---------- Route-based IPsec site-to-site VPN to ASAv (10.10.1.2) ----------
config vpn ipsec phase1-interface
    edit "toASA"
        set interface "port1"
        set ike-version 2
        set peertype any
        set net-device disable
        set proposal aes256-sha256
        set dhgrp 14
        set remote-gw 10.10.1.2
        set psksecret Str0ngVPNkey!
    next
end

config vpn ipsec phase2-interface
    edit "toASA-VLAN10"
        set phase1name "toASA"
        set proposal aes256-sha256
        set src-subnet 172.16.0.0 255.255.0.0
        set dst-subnet 192.168.10.0 255.255.255.0
    next
    edit "toASA-VLAN20"
        set phase1name "toASA"
        set proposal aes256-sha256
        set src-subnet 172.16.0.0 255.255.0.0
        set dst-subnet 192.168.20.0 255.255.255.0
    next
    edit "toASA-VLAN30"
        set phase1name "toASA"
        set proposal aes256-sha256
        set src-subnet 172.16.0.0 255.255.0.0
        set dst-subnet 192.168.30.0 255.255.255.0
    next
end

config router static
    edit 1
        set dst 192.168.10.0 255.255.255.0
        set device "toASA"
    next
    edit 2
        set dst 192.168.20.0 255.255.255.0
        set device "toASA"
    next
    edit 3
        set dst 192.168.30.0 255.255.255.0
        set device "toASA"
    next
end

config firewall address
    edit "SiteB_LAN"
        set subnet 172.16.0.0 255.255.0.0
    next
    edit "SiteA_LAN"
        set subnet 192.168.0.0 255.255.0.0
    next
end

! ---------- Firewall policies ----------
config firewall policy
    edit 1
        set name "LAN-to-VPN"
        set srcintf "port2"
        set dstintf "toASA"
        set srcaddr "SiteB_LAN"
        set dstaddr "SiteA_LAN"
        set action accept
        set schedule "always"
        set service "ALL"
        set utm-status enable
        set ips-sensor "default"
        set av-profile "default"
        set logtraffic all
    next
    edit 2
        set name "VPN-to-LAN"
        set srcintf "toASA"
        set dstintf "port2"
        set srcaddr "SiteA_LAN"
        set dstaddr "SiteB_LAN"
        set action accept
        set schedule "always"
        set service "ALL"
        set logtraffic all
    next
    edit 3
        set name "Block-LAN-to-WAN-default"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "all"
        set dstaddr "all"
        set action deny
        set schedule "always"
        set service "ALL"
        set logtraffic all
    next
end
